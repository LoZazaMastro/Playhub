cmake_minimum_required(VERSION 3.24)
project(PlayhubExplorerFreeDiagnostic LANGUAGES CXX)
option(PLAYHUB_UIACCESS_HOST "Build a separate signing-ready UIAccess binary; never run unsigned" OFF)
set(HOST ExplorerFreeDiagnostic)
set(HOST_MANIFEST ${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
if(PLAYHUB_UIACCESS_HOST)
 set(HOST ExplorerFreeUiAccessDiagnostic)
 set(HOST_MANIFEST ${CMAKE_CURRENT_SOURCE_DIR}/app-uiaccess.manifest)
endif()
add_executable(${HOST} src/main.cpp src/CompositionPresenter.cpp)
set_property(TARGET ${HOST} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(${HOST} PRIVATE cxx_std_20)
target_compile_definitions(${HOST} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(${HOST} PRIVATE /W4 /permissive- /EHsc)
target_link_options(${HOST} PRIVATE /MANIFEST:EMBED /MANIFESTUAC:NO "/MANIFESTINPUT:${HOST_MANIFEST}")
target_link_libraries(${HOST} PRIVATE user32 d3d11 dxgi dcomp dwmapi ole32 advapi32)
