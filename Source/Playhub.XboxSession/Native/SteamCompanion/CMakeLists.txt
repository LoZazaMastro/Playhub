cmake_minimum_required(VERSION 3.20)
project(PlayhubSteamCompanion LANGUAGES CXX)

# Private renderer experiment. Normal companion builds retain the default path.
option(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE "Private renderer texture-source trial" OFF)
if(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../TextureProbe" "texture-probe")
    return()
endif()

add_library(PlayhubSteamCompanion SHARED SteamCompanion.cpp PublicSteamContext.cpp)
target_compile_features(PlayhubSteamCompanion PRIVATE cxx_std_17)
target_compile_definitions(PlayhubSteamCompanion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
option(PLAYHUB_COMPANION_GUIDE_WARMUP "Prepare the overlay on the first verified foreground game" OFF)
option(PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT "Experimental retained-frame overlay attempt for fullscreen games" OFF)
option(PLAYHUB_COMPANION_PUBLIC_CONTEXT_TRIAL "Read public Steam API context in the exact companion; never activate UI" OFF)
set(PLAYHUB_PUBLIC_STEAMCLIENT_SHA256 "" CACHE STRING "Exact Valve-attested installed steamclient64.dll digest for private context trial")
if(PLAYHUB_COMPANION_PUBLIC_CONTEXT_TRIAL)
    string(LENGTH "${PLAYHUB_PUBLIC_STEAMCLIENT_SHA256}" public_digest_length)
    if(NOT public_digest_length EQUAL 64 OR PLAYHUB_PUBLIC_STEAMCLIENT_SHA256 MATCHES "[^0-9A-Fa-f]")
        message(FATAL_ERROR "Public context trial requires the exact 64-digit installed Valve steamclient64.dll SHA256")
    endif()
    target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_COMPANION_PUBLIC_CONTEXT_TRIAL
        PLAYHUB_PUBLIC_STEAMCLIENT_SHA256="${PLAYHUB_PUBLIC_STEAMCLIENT_SHA256}")
endif()
if(PLAYHUB_COMPANION_GUIDE_WARMUP)
    target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_COMPANION_GUIDE_WARMUP)
endif()
if(PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT)
    target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT)
endif()
target_link_libraries(PlayhubSteamCompanion PRIVATE d3d11 dxgi dwmapi user32 gdi32 shell32 ole32 bcrypt)
set_property(TARGET PlayhubSteamCompanion PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(CTest)
if(BUILD_TESTING)
    add_executable(PlayhubSteamPublicContextTests PublicSteamContextTests.cpp)
    target_compile_features(PlayhubSteamPublicContextTests PRIVATE cxx_std_17)
    add_test(NAME steam_public_context_lifetime_policy COMMAND PlayhubSteamPublicContextTests)
    add_executable(PlayhubSteamCompanionPolicyTests PolicyTests.cpp)
    target_compile_features(PlayhubSteamCompanionPolicyTests PRIVATE cxx_std_17)
    add_test(NAME steam_companion_policy COMMAND PlayhubSteamCompanionPolicyTests)
    add_executable(PlayhubSteamCompanionPresentationTests PresentationTests.cpp)
    target_compile_features(PlayhubSteamCompanionPresentationTests PRIVATE cxx_std_17)
    target_compile_definitions(PlayhubSteamCompanionPresentationTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_test(NAME steam_companion_presentation_status COMMAND PlayhubSteamCompanionPresentationTests)
    add_executable(PlayhubSteamCompanionSnapshotTests SnapshotTests.cpp)
    target_compile_features(PlayhubSteamCompanionSnapshotTests PRIVATE cxx_std_17)
    target_compile_definitions(PlayhubSteamCompanionSnapshotTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PlayhubSteamCompanionSnapshotTests PRIVATE gdi32 user32)
    add_test(NAME steam_companion_snapshot_pixels COMMAND PlayhubSteamCompanionSnapshotTests)
    add_executable(PlayhubSteamCompanionSnapshotBlitTests SnapshotBlitTests.cpp)
    target_compile_features(PlayhubSteamCompanionSnapshotBlitTests PRIVATE cxx_std_17)
    target_compile_definitions(PlayhubSteamCompanionSnapshotBlitTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PlayhubSteamCompanionSnapshotBlitTests PRIVATE d3d11)
    add_test(NAME steam_companion_snapshot_warp_scaling COMMAND PlayhubSteamCompanionSnapshotBlitTests)
endif()
