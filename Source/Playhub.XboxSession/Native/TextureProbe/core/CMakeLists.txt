cmake_minimum_required(VERSION 3.20)
project(PlayhubValveForegroundView LANGUAGES C CXX)
add_library(PrivateMinHook STATIC MinHook/src/buffer.c MinHook/src/hook.c MinHook/src/trampoline.c MinHook/src/hde/hde64.c)
target_include_directories(PrivateMinHook PUBLIC MinHook/include)
add_library(ValveForegroundView STATIC CallerImage.cpp RendererAttestation.cpp ScopedForegroundView.cpp)
target_compile_features(ValveForegroundView PUBLIC cxx_std_17)
if(PLAYHUB_VALVE_ACTIVE_WINDOW_VIEW)
 target_compile_definitions(ValveForegroundView PRIVATE PLAYHUB_VALVE_ACTIVE_WINDOW_VIEW=1)
endif()
if(PLAYHUB_VALVE_WINDOW_OBSERVATION)
 target_compile_definitions(ValveForegroundView PRIVATE PLAYHUB_VALVE_WINDOW_OBSERVATION=1)
endif()
target_compile_definitions(ValveForegroundView PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ValveForegroundView PUBLIC PrivateMinHook user32 advapi32 bcrypt wintrust crypt32 shell32 ole32 uuid version)
add_library(RendererFixture SHARED Fixture.cpp)
add_library(ForeignFixture SHARED Fixture.cpp)
add_executable(PolicyTests Tests.cpp)
target_compile_features(PolicyTests PRIVATE cxx_std_17)
target_link_libraries(PolicyTests PRIVATE ValveForegroundView)
add_dependencies(PolicyTests RendererFixture ForeignFixture)
foreach(target PrivateMinHook ValveForegroundView RendererFixture ForeignFixture PolicyTests)
 set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
# The fixture must call (not tail-jump) its function pointer so _ReturnAddress
# identifies the real DLL call site. It never creates a thread or GUI object.
target_compile_options(RendererFixture PRIVATE /Od /Ob0)
target_compile_options(ForeignFixture PRIVATE /Od /Ob0)
enable_testing()
add_test(NAME pure_policy_only COMMAND PolicyTests --pure)
add_test(NAME own_inert_native_caller_only COMMAND PolicyTests --native-fixture)

add_executable(ObservationTraceTests ObservationTraceTests.cpp)
target_compile_features(ObservationTraceTests PRIVATE cxx_std_17)
set_property(TARGET ObservationTraceTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME pure_window_observation_policy COMMAND ObservationTraceTests)

add_executable(ActiveViewChecks ActiveViewChecks.cpp)
target_compile_features(ActiveViewChecks PRIVATE cxx_std_17)
target_compile_definitions(ActiveViewChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX PLAYHUB_VALVE_ACTIVE_WINDOW_VIEW=1)
target_include_directories(ActiveViewChecks PRIVATE MinHook/include)
set_property(TARGET ActiveViewChecks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME actual_null_active_view_fake_api_only COMMAND ActiveViewChecks)

add_executable(RendererAdmissionChecks RendererAdmissionChecks.cpp)
target_compile_features(RendererAdmissionChecks PRIVATE cxx_std_17)
target_compile_definitions(RendererAdmissionChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(RendererAdmissionChecks PRIVATE user32 advapi32 bcrypt wintrust crypt32 version)
set_property(TARGET RendererAdmissionChecks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME retained_renderer_attestation_native_read_only COMMAND RendererAdmissionChecks)
