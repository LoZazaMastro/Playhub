cmake_minimum_required(VERSION 3.20)
project(PlayhubRendererTextureAdapter LANGUAGES C CXX)
option(PLAYHUB_VALVE_WINDOW_OBSERVATION "Private transparent Valve5API diagnostic only" OFF)
option(PLAYHUB_VALVE_ACTIVE_WINDOW_VIEW "Private NULL-only own-thread GetActiveWindow view" OFF)
if(PLAYHUB_VALVE_ACTIVE_WINDOW_VIEW AND NOT PLAYHUB_VALVE_WINDOW_OBSERVATION)
 message(FATAL_ERROR "NULL-only active window view requires explicit observation instrumentation")
endif()
option(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE "Private render-only companion path" OFF)
option(PLAYHUB_COMPANION_LIVE_MIRROR "Private owned live GPU mirror path; managed configuration required" OFF)
set(PLAYHUB_MIRROR_COMPONENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../OverlayMirror" CACHE PATH "Reviewed complete mirror component preferred source")
option(PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT "Retain original fullscreen eligibility policy; private path does not capture" OFF)
option(PLAYHUB_COMPANION_PUBLIC_CONTEXT_TRIAL "Public SDK trial (excluded from texture probe)" OFF)
option(PLAYHUB_TEXTURE_RENDERER_ATTESTED "Admit only the retained registered Valve Steam renderer by runtime file/module contract" OFF)
if(PLAYHUB_TEXTURE_RENDERER_ATTESTED AND NOT PLAYHUB_COMPANION_LIVE_MIRROR)
 message(FATAL_ERROR "Attested Steam renderer is limited to the private live mirror")
endif()
set(PLAYHUB_TEXTURE_RENDERER_SHA256 "" CACHE STRING "Exact installed Valve renderer digest")
if(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE OR PLAYHUB_COMPANION_LIVE_MIRROR)
 if(PLAYHUB_TEXTURE_RENDERER_ATTESTED)
  if(NOT PLAYHUB_COMPANION_LIVE_MIRROR OR PLAYHUB_TEXTURE_RENDERER_SHA256)
   message(FATAL_ERROR "Attested Steam renderer requires live mirror and excludes fixed-digest mode")
  endif()
 else()
  string(LENGTH "${PLAYHUB_TEXTURE_RENDERER_SHA256}" digest_length)
  if(NOT digest_length EQUAL 64 OR PLAYHUB_TEXTURE_RENDERER_SHA256 MATCHES "[^0-9A-Fa-f]")
   message(FATAL_ERROR "Private texture probe requires an exact renderer SHA256")
  endif()
 endif()
 if(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE AND PLAYHUB_COMPANION_LIVE_MIRROR)
  message(FATAL_ERROR "Select the live mirror or the historical A/B probe, never both")
 endif()
 if(PLAYHUB_COMPANION_PUBLIC_CONTEXT_TRIAL)
  message(FATAL_ERROR "Texture probe excludes the unrelated public Steam SDK trial")
 endif()
endif()
add_subdirectory(core core-build)
enable_testing()
add_executable(TextureProbePolicyTests native/TextureProbeTests.cpp)
target_compile_features(TextureProbePolicyTests PRIVATE cxx_std_17)
set_property(TARGET TextureProbePolicyTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME texture_probe_policy_only COMMAND TextureProbePolicyTests)
add_library(PlayhubSteamCompanion SHARED native/SteamCompanion.cpp native/PublicSteamContext.cpp)
target_compile_features(PlayhubSteamCompanion PRIVATE cxx_std_17)
target_compile_definitions(PlayhubSteamCompanion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(PlayhubSteamCompanion PRIVATE core)
target_link_libraries(PlayhubSteamCompanion PRIVATE ValveForegroundView d3d11 dxgi dwmapi user32 gdi32 shell32 ole32 bcrypt)
if(PLAYHUB_COMPANION_LIVE_MIRROR)
 add_subdirectory("${PLAYHUB_MIRROR_COMPONENT_ROOT}" overlay-mirror-build)
 target_link_libraries(PlayhubSteamCompanion PRIVATE PlayhubOverlayMirror)
 target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_COMPANION_LIVE_MIRROR PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE)
 if(PLAYHUB_TEXTURE_RENDERER_ATTESTED)
  target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_TEXTURE_RENDERER_ATTESTED=1)
 else()
  target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_TEXTURE_RENDERER_SHA256="${PLAYHUB_TEXTURE_RENDERER_SHA256}")
 endif()
 add_executable(LiveMirrorCallsiteTests native/LiveMirrorCallsiteTests.cpp)
 target_include_directories(LiveMirrorCallsiteTests PRIVATE "${PLAYHUB_MIRROR_COMPONENT_ROOT}/native" core)
 target_compile_features(LiveMirrorCallsiteTests PRIVATE cxx_std_17)
 set_property(TARGET LiveMirrorCallsiteTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_test(NAME live_mirror_callsite_api_and_renewal_no_os COMMAND LiveMirrorCallsiteTests)
endif()
set_property(TARGET PlayhubSteamCompanion PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE)
 target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_FOREGROUND_VIEW_TEXTURE_PROBE PLAYHUB_TEXTURE_RENDERER_SHA256="${PLAYHUB_TEXTURE_RENDERER_SHA256}")
endif()
if(PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT)
 target_compile_definitions(PlayhubSteamCompanion PRIVATE PLAYHUB_COMPANION_FULLSCREEN_SNAPSHOT)
endif()
