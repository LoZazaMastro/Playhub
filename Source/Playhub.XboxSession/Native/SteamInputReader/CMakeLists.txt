cmake_minimum_required(VERSION 3.20)
project(PlayhubSteamInputReader LANGUAGES CXX)
add_executable(PlayhubSteamInputReader SteamInputReader.cpp)
target_compile_features(PlayhubSteamInputReader PRIVATE cxx_std_20)
target_compile_definitions(PlayhubSteamInputReader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PlayhubSteamInputReader PRIVATE user32 hid xinput)
if(MSVC)
    set_property(TARGET PlayhubSteamInputReader PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(PlayhubSteamInputReader PRIVATE /W4 /WX)
    target_link_options(PlayhubSteamInputReader PRIVATE /DEPENDENTLOADFLAG:0x800)
endif()
