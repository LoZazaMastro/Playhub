cmake_minimum_required(VERSION 3.20)
project(PlayhubOverlayMirrorComponent LANGUAGES CXX)
# A reviewed normal candidate may reuse its exact attested callback artifact.
# Metadata must describe the SAME callback source; this never changes runtime trust.
set(PLAYHUB_MIRROR_CALLBACK_FILE "" CACHE FILEPATH "Optional exact reviewed callback")
set(PLAYHUB_MIRROR_CALLBACK_MANIFEST "" CACHE FILEPATH "Required source/hash metadata for explicit callback")
if(PLAYHUB_MIRROR_CALLBACK_FILE)
 get_filename_component(callback_name "${PLAYHUB_MIRROR_CALLBACK_FILE}" NAME)
 if(NOT callback_name STREQUAL "PlayhubExactBand.dll" OR NOT EXISTS "${PLAYHUB_MIRROR_CALLBACK_FILE}" OR IS_DIRECTORY "${PLAYHUB_MIRROR_CALLBACK_FILE}" OR NOT EXISTS "${PLAYHUB_MIRROR_CALLBACK_MANIFEST}")
  message(FATAL_ERROR "Reviewed callback artifact or metadata is absent/invalid")
 endif()
 file(READ "${PLAYHUB_MIRROR_CALLBACK_MANIFEST}" callback_metadata)
 string(JSON callback_protocol GET "${callback_metadata}" protocol)
 string(JSON expected_callback_sha GET "${callback_metadata}" callbackSha256)
 string(JSON expected_source_digest GET "${callback_metadata}" sourceDigest)
 file(SHA256 "${PLAYHUB_MIRROR_CALLBACK_FILE}" actual_callback_sha)
 string(TOUPPER "${actual_callback_sha}" actual_callback_sha)
 if(NOT callback_protocol EQUAL 1 OR NOT actual_callback_sha STREQUAL expected_callback_sha)
  message(FATAL_ERROR "Reviewed callback hash/protocol mismatch")
 endif()
 set(callback_source_names CMakeLists.txt Hook.cpp Identity.h Policy.h Protocol.h ReadDiagnostic.h Target.h)
 string(JSON callback_source_count LENGTH "${callback_metadata}" sources)
 list(LENGTH callback_source_names callback_expected_count)
 if(NOT callback_source_count EQUAL callback_expected_count)
  message(FATAL_ERROR "Reviewed callback source count mismatch")
 endif()
 set(actual_source_record "")
 foreach(name IN LISTS callback_source_names)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/callback-v3/${name}" source_sha)
  string(TOUPPER "${source_sha}" source_sha)
  string(JSON expected_source_sha GET "${callback_metadata}" sources "${name}")
  if(NOT source_sha STREQUAL expected_source_sha)
   message(FATAL_ERROR "Reviewed callback source mismatch: ${name}")
  endif()
  string(APPEND actual_source_record "${name}:${source_sha}\n")
 endforeach()
 string(SHA256 actual_source_digest "${actual_source_record}")
 string(TOUPPER "${actual_source_digest}" actual_source_digest)
 if(NOT actual_source_digest STREQUAL expected_source_digest)
  message(FATAL_ERROR "Reviewed callback source digest mismatch")
 endif()
 add_library(PlayhubExactBand SHARED IMPORTED GLOBAL)
 set_target_properties(PlayhubExactBand PROPERTIES IMPORTED_LOCATION "${PLAYHUB_MIRROR_CALLBACK_FILE}")
 set(callback_artifact_dependency "${PLAYHUB_MIRROR_CALLBACK_FILE}" "${PLAYHUB_MIRROR_CALLBACK_MANIFEST}")
else()
 add_subdirectory(callback-v3)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/ExpectedBandCallback.h"
 COMMAND "${CMAKE_COMMAND}" -DDLL=$<TARGET_FILE:PlayhubExactBand> -DHEADER=${CMAKE_CURRENT_BINARY_DIR}/generated/ExpectedBandCallback.h -P "${CMAKE_CURRENT_SOURCE_DIR}/HashCallback.cmake"
 DEPENDS PlayhubExactBand ${callback_artifact_dependency} "${CMAKE_CURRENT_SOURCE_DIR}/HashCallback.cmake" VERBATIM)
add_library(PlayhubOverlayMirror STATIC native/Win32MirrorBackend.cpp native/C449Band16Broker.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/ExpectedBandCallback.h")
target_compile_features(PlayhubOverlayMirror PUBLIC cxx_std_17)
target_include_directories(PlayhubOverlayMirror PUBLIC native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(PlayhubOverlayMirror PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PlayhubOverlayMirror PUBLIC user32 gdi32 advapi32 dwmapi bcrypt psapi wintrust crypt32 shell32 ole32 uuid)
set_property(TARGET PlayhubOverlayMirror PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(MirrorPolicyTests native/MirrorPolicyTests.cpp)
target_compile_features(MirrorPolicyTests PRIVATE cxx_std_17)
set_property(TARGET MirrorPolicyTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME mirror_lifecycle_no_windows COMMAND MirrorPolicyTests)

add_executable(BandBrokerTests native/BandBrokerTests.cpp)
target_compile_features(BandBrokerTests PRIVATE cxx_std_17)
target_compile_definitions(BandBrokerTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET BandBrokerTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME mirror_broker_wire_and_retirement_no_os COMMAND BandBrokerTests)

add_executable(MirrorLinkTests native/MirrorLinkTests.cpp)
target_link_libraries(MirrorLinkTests PRIVATE PlayhubOverlayMirror)
set_property(TARGET MirrorLinkTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME mirror_complete_link_null_factories_no_os COMMAND MirrorLinkTests)
