cmake_minimum_required(VERSION 3.20)
project(PlayhubHudDisplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(HudDisplayChecks tests/HudDisplayChecks.cpp)
target_include_directories(HudDisplayChecks PRIVATE native)
if(MSVC)
 target_compile_options(HudDisplayChecks PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME hud_display_only_lease_no_os COMMAND HudDisplayChecks)

add_subdirectory(OverlayMirror)
add_library(PlayhubHudDisplay STATIC native/HudWindowsBackend.cpp)
target_link_libraries(PlayhubHudDisplay PUBLIC PlayhubOverlayMirror)
target_include_directories(PlayhubHudDisplay PUBLIC native)
target_compile_definitions(PlayhubHudDisplay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(PlayhubHudDisplayNative SHARED native/HudDisplayApi.cpp)
target_link_libraries(PlayhubHudDisplayNative PRIVATE PlayhubHudDisplay)
add_executable(HudCallbackChecks tests/HudCallbackChecks.generated.cpp)
target_link_libraries(HudCallbackChecks PRIVATE PlayhubOverlayMirror)
add_test(NAME hud_whole_callback_inert_native_apis COMMAND HudCallbackChecks)
add_executable(HudBackendChecks tests/HudBackendChecks.generated.cpp)
target_link_libraries(HudBackendChecks PRIVATE PlayhubHudDisplay)
add_test(NAME hud_actual_backend_commit_boundaries_inert COMMAND HudBackendChecks)
add_executable(HudMailboxChecks tests/HudMailboxChecks.cpp)
target_include_directories(HudMailboxChecks PRIVATE native)
add_test(NAME hud_mailbox_authority_races_no_os COMMAND HudMailboxChecks)

# Separate purpose: original LC Main artwork, exact foreground game, first observed content only.
add_library(PlayhubCurtainDisplayNative SHARED native/CurtainDisplayApi.cpp native/CurtainWindowsBackend.cpp)
target_link_libraries(PlayhubCurtainDisplayNative PRIVATE PlayhubOverlayMirror)
target_include_directories(PlayhubCurtainDisplayNative PRIVATE native)
target_compile_definitions(PlayhubCurtainDisplayNative PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
 target_compile_options(PlayhubCurtainDisplayNative PRIVATE /W4 /WX)
endif()

foreach(name CurtainBrokerChecks CurtainDisplayChecks CurtainBackendChecks CurtainMailboxChecks)
 add_executable(${name} tests/${name}.cpp)
 target_include_directories(${name} PRIVATE native OverlayMirror/native OverlayMirror/callback-v3)
 target_compile_definitions(${name} PRIVATE UNICODE _UNICODE NOMINMAX)
 if(MSVC)
  target_compile_options(${name} PRIVATE /W4 /WX)
 endif()
endforeach()
target_link_libraries(CurtainBackendChecks PRIVATE user32 gdi32 dwmapi advapi32 shell32 propsys ole32)
add_test(NAME curtain_authority_fake COMMAND CurtainBrokerChecks)
add_test(NAME curtain_sequence_fake COMMAND CurtainDisplayChecks)
add_test(NAME curtain_backend_fake COMMAND CurtainBackendChecks)
add_test(NAME curtain_mailbox_fake COMMAND CurtainMailboxChecks)
